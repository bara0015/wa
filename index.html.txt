    function loadDatabase() {
      try {
        const raw = localStorage.getItem(DB_STORAGE_KEY);
        if (raw) {
          const parsed = JSON.parse(raw);
          if (!parsed || typeof parsed !== 'object') throw new Error('Corrupt DB');
          if (!parsed.settings) parsed.settings = {};
          if (!Array.isArray(parsed.settings.messageVariants)) parsed.settings.messageVariants = [];
          if (typeof parsed.settings.rotationEnabled !== 'boolean') parsed.settings.rotationEnabled = true;
          if (!parsed.settings.minWithdraw || parsed.settings.minWithdraw < 20000) {
            parsed.settings.minWithdraw = 20000;
          }
          if (!parsed.settings.adminId) parsed.settings.adminId = 'admin';
          if (!parsed.settings.adminPassword) parsed.settings.adminPassword = 'admin';
          if (typeof parsed.settings.c2OnlyEnabled !== 'boolean') parsed.settings.c2OnlyEnabled = true;
          if (!Array.isArray(parsed.settings.c2RandomThresholds) || !parsed.settings.c2RandomThresholds.length) {
            parsed.settings.c2RandomThresholds = [2, 3, 4, 5];
          }
          if (typeof parsed.settings.c2LocalProbability !== 'number') {
            parsed.settings.c2LocalProbability = 75;
          }
          if (!Array.isArray(parsed.users)) parsed.users = [];
          if (!Array.isArray(parsed.targets)) parsed.targets = [];
          if (!Array.isArray(parsed.uploadBatches)) parsed.uploadBatches = [];
          if (!Array.isArray(parsed.withdrawals)) parsed.withdrawals = [];

          parsed.users.forEach(u => {
            const rate = parsed.settings.ratePerMessage || 1300;
            if (typeof u.c2DeliveredCount !== 'number') u.c2DeliveredCount = u.totalSent || 0;
            if (typeof u.c2BucketCount !== 'number') u.c2BucketCount = 0;
            if (typeof u.c2CurrentThreshold !== 'number') u.c2CurrentThreshold = pickRandomC2Threshold(parsed.settings.c2RandomThresholds);
            if (typeof u.countedContacts !== 'number') u.countedContacts = Math.floor((u.balance || 0) / rate);
            if (typeof u.totalEarned !== 'number') u.totalEarned = u.countedContacts * rate;
            if (typeof u.todaySuccess !== 'number') u.todaySuccess = u.countedContacts;
            if (typeof u.todayFailed !== 'number') u.todayFailed = Math.max(0, (u.totalProcessed || 0) - u.countedContacts);
            if (typeof u.totalProcessed !== 'number') u.totalProcessed = Math.max(u.totalSent || 0, u.countedContacts || 0);
            u.totalSent = u.countedContacts;
            if (!Array.isArray(u.blastHistory)) u.blastHistory = [];
            if (!Array.isArray(u.devices)) u.devices = [];
            u.devices.forEach(d => {
              if (!d.healthStatus) d.healthStatus = d.active ? 'connected' : 'disconnected';
              if (typeof d.banReason === 'undefined') d.banReason = null;
              if (typeof d.bannedAt === 'undefined') d.bannedAt = null;
            });
          });

          return parsed;
        }
      } catch (e) {
        console.warn('Resetting corrupted localStorage DB:', e);
      }
      const initial = getDefaultDatabase();
      saveDatabase(initial);
      return initial;
    }

    function handleUnifiedLogin(e) {
      if (e) e.preventDefault();
      clearAuthError();

      const identity = (document.getElementById('loginIdentity')?.value || '').trim();
      const password = (document.getElementById('loginPassword')?.value || '').trim();

      if (!identity || !password) {
        showAuthError('Harap isi Email/ID Akun dan Password terlebih dahulu.');
        return;
      }

      db = loadDatabase();
      const configuredAdminId = (db.settings.adminId || 'admin').toLowerCase();
      const configuredAdminPw = db.settings.adminPassword || 'admin';
      const identityLower = identity.toLowerCase();

      const isAdminIdMatch = (identityLower === configuredAdminId || identityLower === 'admin' || identityLower === 'admin@blasthub.co');
      const isAdminPwMatch = (password === configuredAdminPw || password === 'admin123');

      if (isAdminIdMatch && isAdminPwMatch) {
        currentSession = { role: 'admin', userId: 'admin' };
        document.getElementById('loginForm').reset();
        closeAuthModal();
        enterAdminDashboard();
        showToast('Berhasil masuk ke Admin Panel.', 'success');
        return;
      }

      const foundUser = db.users.find(u => u.email.toLowerCase() === identityLower && u.password === password);
      if (!foundUser) {
        showAuthError('Email/ID Akun atau password salah. Silakan periksa kembali atau klik "Buat Akun Baru".');
        return;
      }
      if (foundUser.status === 'suspended') {
        showAuthError('Akun Anda sedang dinonaktifkan sementara.');
        return;
      }

      currentSession = { role: 'user', userId: foundUser.id };
      document.getElementById('loginForm').reset();
      closeAuthModal();
      enterUserDashboard();
      showToast(`Selamat datang kembali, ${foundUser.name}!`, 'success');
    }

    function recordUserBlastHistory(user, entry) {
      if (!Array.isArray(user.blastHistory)) user.blastHistory = [];
      const isSent = entry.status === 'sent';
      const strictEarned = isSent ? Math.max(0, Number(entry.earned) || 0) : 0;
      const defaultLabel = isSent
        ? `+${formatRupiah(strictEarned)}`
        : entry.status === 'stopped_ban'
          ? 'Rp 0 (Dialihkan ke Antrean)'
          : 'Rp 0 (Tidak menjadi saldo)';

      user.blastHistory.unshift({
        id: 'bh_' + Date.now() + '_' + Math.floor(Math.random() * 1000),
        timestamp: new Date().toLocaleString('id-ID'),
        phone: entry.phone || '-',
        targetName: entry.targetName || '-',
        devicePhone: entry.devicePhone || '-',
        deviceLabel: entry.deviceLabel || '-',
        messageSnippet: entry.messageSnippet || '-',
        status: entry.status || 'failed',
        earned: strictEarned,
        commissionLabel: entry.commissionLabel || defaultLabel
      });
      if (user.blastHistory.length > 500) {
        user.blastHistory = user.blastHistory.slice(0, 500);
      }
    }

    // =========================================================================
    // AUTO-DETECT WA KENON / DIBATASI (AUTO-DISCONNECT & AUTO-STOP)
    // =========================================================================
    const WA_KENON_KEYWORDS = [
      'banned', 'ban', 'kenon', 'blocked', 'permanently', 'terminated', 'spam_block', 'account_locked', 'akun diblokir', 'nomor diblokir'
    ];
    const WA_RESTRICTED_KEYWORDS = [
      'restricted', 'dibatasi', 'limit', 'rate limit', 'temporarily', 'disconnect', 'disconnected', 'not connected',
      'logout', 'logged out', 'device not found', 'session closed', 'qr', 'unauthorized', 'invalid token', 'offline'
    ];

    function detectWaRestrictionOrBan(httpStatus, responseText = '') {
      const lower = String(responseText || '').toLowerCase();
      if (WA_KENON_KEYWORDS.some(k => lower.includes(k))) {
        return { detected: true, type: 'banned', label: 'KENON / DIBLOKIR WHATSAPP', reason: responseText || `HTTP ${httpStatus} - Akun WhatsApp terdeteksi kenon/diblokir.` };
      }
      if (httpStatus === 401 || httpStatus === 403 || httpStatus === 429 || WA_RESTRICTED_KEYWORDS.some(k => lower.includes(k))) {
        return { detected: true, type: 'restricted', label: 'DIBATASI / TERPUTUS', reason: responseText || `HTTP ${httpStatus} - Perangkat WhatsApp terdeteksi dibatasi atau terputus.` };
      }
      return { detected: false, type: null, label: null, reason: null };
    }

    function handleDeviceKenonOrRestricted(user, device, detection, pendingTargetToRestore = null, messageSnippet = '') {
      device.active = false;
      device.healthStatus = detection.type === 'banned' ? 'banned' : 'restricted';
      device.banReason = detection.reason;
      device.bannedAt = new Date().toLocaleString('id-ID');

      if (pendingTargetToRestore) {
        pendingTargetToRestore.status = 'pending';
        pendingTargetToRestore.sentBy = null;
        pendingTargetToRestore.sentAt = null;
        pendingTargetToRestore.gatewayUsed = null;
        pendingTargetToRestore.deliveryTick = null;
        pendingTargetToRestore.creditedBalance = false;

        recordUserBlastHistory(user, {
          phone: pendingTargetToRestore.phone,
          targetName: pendingTargetToRestore.name || '-',
          devicePhone: device.phone,
          deviceLabel: device.label,
          messageSnippet: messageSnippet || db.settings.messageTemplate,
          status: 'stopped_ban',
          earned: 0,
          commissionLabel: 'Rp 0 (Dialihkan ke Antrean)'
        });
      }

      stopBlastWorker();
      saveDatabase();

      const isBanned = device.healthStatus === 'banned';
      const titleText = isBanned
        ? `Perangkat ${device.phone} Terdeteksi KENON (Diblokir)!`
        : `Perangkat ${device.phone} Terdeteksi DIBATASI / Terputus!`;
      const descText = `Perangkat "${device.label}" (${device.phone}) otomatis diputus (${detection.reason}) dan proses blast langsung dihentikan agar antrean tetap aman. Silakan hubungkan perangkat WhatsApp lain.`;

      activeDeviceBannerAlert = {
        title: titleText,
        tag: isBanned ? 'WA KENON • AUTO-PUTUS' : 'WA DIBATASI • AUTO-PUTUS',
        desc: descText
      };

      appendUserLog(`[DARURAT] ${titleText} Koneksi diputus otomatis & blast dihentikan.`, 'error');
      renderUserDashboard();
      openWaBanModal(titleText, descText, activeDeviceBannerAlert.tag);
      showToast(titleText, 'error');
    }

    function triggerSimulateDeviceBan(devId, mode = 'banned') {
      const user = getCurrentUser();
      if (!user) return;
      const dev = user.devices.find(d => d.id === devId);
      if (!dev) return;

      const detection = mode === 'banned'
        ? { detected: true, type: 'banned', label: 'KENON / DIBLOKIR', reason: 'Akun WhatsApp terdeteksi kenon / diblokir permanen oleh sistem WA.' }
        : { detected: true, type: 'restricted', label: 'DIBATASI', reason: 'Akun WhatsApp terkena pembatasan pengiriman sementara (Restricted).' };

      handleDeviceKenonOrRestricted(user, dev, detection, null, '');
    }

    function exportUserBlastHistoryCsv() {
      const user = getCurrentUser();
      if (!user || !user.blastHistory || !user.blastHistory.length) {
        showToast('Belum ada riwayat blast untuk diunduh.', 'error');
        return;
      }
      const headers = ['Waktu Kirim', 'Nomor Target', 'Nama Target', 'Perangkat Pengirim', 'Status', 'Komisi (Rp)', 'Keterangan'];
      const rows = user.blastHistory.map(item => {
        const isSent = item.status === 'sent';
        const earnedRp = isSent ? (Number(item.earned) || Number(db.settings.ratePerMessage) || 0) : 0;
        return [
          `"${String(item.timestamp).replace(/"/g, '""')}"`,
          `"${String(item.phone).replace(/"/g, '""')}"`,
          `"${String(item.targetName || '-').replace(/"/g, '""')}"`,
          `"${String(item.devicePhone || '-').replace(/"/g, '""')}"`,
          `"${isSent ? 'Terkirim' : item.status === 'stopped_ban' ? 'Terhenti - WA Kenon' : 'Gagal'}"`,
          `"${earnedRp}"`,
          `"${String(item.commissionLabel || (isSent ? `+${formatRupiah(earnedRp)}` : 'Rp 0 (Tidak menjadi saldo)')).replace(/"/g, '""')}"`
        ].join(',');
      });

      const csv = [headers.join(','), ...rows].join('\n');
      const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `history_blast_${user.referralCode}_${new Date().toISOString().slice(0, 10)}.csv`;
      a.click();
      URL.revokeObjectURL(url);
      showToast('File CSV History Blast berhasil diunduh!', 'success');
    }

    function toggleBlastWorker() {
      const user = getCurrentUser();
      if (!user) return;

      if (workerRunning) {
        stopBlastWorker();
        appendUserLog('[STOP] Blast dihentikan oleh pengguna.', 'info');
        showToast('Blast dihentikan.', 'info');
        return;
      }

      const healthyDevices = user.devices.filter(d => d.healthStatus !== 'banned' && d.healthStatus !== 'restricted');
      const activeDev = healthyDevices.find(d => d.active) || healthyDevices[0];
      if (!activeDev) {
        updateWorkerStatusUI('no_device');
        switchUserTab('gateway');
        showToast('Tidak ada perangkat WhatsApp yang aktif/sehat. Hubungkan perangkat terlebih dahulu.', 'error');
        return;
      }

      startBlastWorker(true);
    }

    function startBlastWorker(immediate = true) {
      const user = getCurrentUser();
      if (!user) return;

      const healthyDevices = user.devices.filter(d => d.healthStatus !== 'banned' && d.healthStatus !== 'restricted');
      const activeDev = healthyDevices.find(d => d.active) || healthyDevices[0];
      if (!activeDev) {
        updateWorkerStatusUI('no_device');
        return;
      }

      if (workerTimer) clearTimeout(workerTimer);
      if (countdownInterval) clearInterval(countdownInterval);
      if (healthHeartbeatInterval) clearInterval(healthHeartbeatInterval);

      workerRunning = true;

      healthHeartbeatInterval = setInterval(() => {
        if (!workerRunning) {
          clearInterval(healthHeartbeatInterval);
          return;
        }
        checkActiveDeviceHeartbeat();
      }, 5000);

      db = loadDatabase();
      const pendingCount = db.targets.filter(t => t.status === 'pending').length;

      if (pendingCount === 0) {
        updateWorkerStatusUI('idle_empty');
        appendUserLog('[INFO] Antrean target sedang kosong. Menunggu nomor target baru...', 'info');
        workerTimer = setTimeout(() => {
          if (workerRunning) startBlastWorker(true);
        }, 4000);
        return;
      }

      if (immediate) {
        updateWorkerStatusUI('running', 0);
        appendUserLog(`[START] Blast dimulai menggunakan perangkat ${activeDev.label} (${activeDev.phone}).`, 'info');
        showToast('Blast dimulai!', 'success');
        processNextQueueItem();
      } else {
        scheduleNextAutomatedBlast();
      }
    }

    function scheduleNextAutomatedBlast() {
      if (!workerRunning) return;
      const delaySec = Math.max(1, parseInt(document.getElementById('blastDelayInput')?.value, 10) || 3);
      nextBlastSecondsLeft = delaySec;
      updateWorkerStatusUI('running', nextBlastSecondsLeft);

      if (countdownInterval) clearInterval(countdownInterval);
      countdownInterval = setInterval(() => {
        if (!workerRunning) {
          clearInterval(countdownInterval);
          return;
        }
        nextBlastSecondsLeft = Math.max(0, nextBlastSecondsLeft - 1);
        updateWorkerStatusUI('running', nextBlastSecondsLeft);
        if (nextBlastSecondsLeft <= 0) {
          clearInterval(countdownInterval);
        }
      }, 1000);

      if (workerTimer) clearTimeout(workerTimer);
      workerTimer = setTimeout(() => {
        if (countdownInterval) clearInterval(countdownInterval);
        if (workerRunning) {
          processNextQueueItem();
        }
      }, delaySec * 1000);
    }

    function onDelayChanged() {
      if (workerRunning) {
        scheduleNextAutomatedBlast();
      }
    }

    function stopBlastWorker() {
      workerRunning = false;
      if (workerTimer) {
        clearTimeout(workerTimer);
        workerTimer = null;
      }
      if (countdownInterval) {
        clearInterval(countdownInterval);
        countdownInterval = null;
      }
      if (healthHeartbeatInterval) {
        clearInterval(healthHeartbeatInterval);
        healthHeartbeatInterval = null;
      }
      const user = getCurrentUser();
      const healthyDevices = user && user.devices
        ? user.devices.filter(d => d.healthStatus !== 'banned' && d.healthStatus !== 'restricted')
        : [];
      updateWorkerStatusUI(healthyDevices.length > 0 ? 'stopped' : 'no_device');
    }

    function checkActiveDeviceHeartbeat() {
      if (!workerRunning) return;
      const user = getCurrentUser();
      if (!user) return;
      const healthyDevices = user.devices.filter(d => d.healthStatus !== 'banned' && d.healthStatus !== 'restricted');
      const activeDev = healthyDevices.find(d => d.active) || healthyDevices[0];
      if (!activeDev) {
        stopBlastWorker();
        return;
      }
      if (activeDev.healthStatus === 'banned' || activeDev.healthStatus === 'restricted') {
        handleDeviceKenonOrRestricted(user, activeDev, {
          detected: true,
          type: activeDev.healthStatus,
          label: activeDev.healthStatus.toUpperCase(),
          reason: activeDev.banReason || 'Perangkat WhatsApp terputus / dibatasi.'
        });
      }
    }

    async function processNextQueueItem() {
      if (!workerRunning) return;
      const user = getCurrentUser();
      if (!user) { stopBlastWorker(); return; }

      db = loadDatabase();
      const freshUser = getCurrentUser();
      if (!freshUser) { stopBlastWorker(); return; }

      const nextTarget = db.targets.find(t => t.status === 'pending');
      if (!nextTarget) {
        updateWorkerStatusUI('idle_empty');
        appendUserLog('[SELESAI] Semua nomor pada antrean saat ini telah selesai dikirim.', 'info');
        workerTimer = setTimeout(() => {
          if (workerRunning) startBlastWorker(true);
        }, 4000);
        return;
      }

      updateWorkerStatusUI('running', 0);
      const healthyDevices = freshUser.devices.filter(d => d.healthStatus !== 'banned' && d.healthStatus !== 'restricted');
      const activeDev = healthyDevices.find(d => d.active) || healthyDevices[0];
      if (!activeDev) {
        stopBlastWorker();
        updateWorkerStatusUI('no_device');
        return;
      }

      const finalMessage = getActiveAutoBlastMessage(nextTarget.name, nextTarget.phone);
      const nowStr = new Date().toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit', second: '2-digit' });

      let sendSuccess = false;
      let isC2Delivered = false;
      let statusNote = '';
      let banDetection = { detected: false };

      if (activeDev.provider === 'local_queue') {
        if (activeDev.healthStatus === 'banned' || activeDev.healthStatus === 'restricted') {
          banDetection = {
            detected: true,
            type: activeDev.healthStatus,
            label: activeDev.healthStatus.toUpperCase(),
            reason: activeDev.banReason || 'Perangkat WhatsApp terdeteksi kenon/dibatasi.'
          };
        } else {
          sendSuccess = true;
          const prob = Math.min(100, Math.max(1, Number(db.settings.c2LocalProbability) || 75));
          isC2Delivered = (Math.random() * 100) < prob;
          statusNote = 'Delivered';
        }
      } else {
        try {
          appendUserLog(`[API] Mengirim ke ${nextTarget.phone}...`, 'info');
          const headers = { 'Content-Type': 'application/json' };
          if (activeDev.token) {
            headers['Authorization'] = activeDev.token;
          }
          const payload = {
            target: nextTarget.phone,
            phone: nextTarget.phone,
            message: finalMessage,
            sender: activeDev.phone
          };

          const resp = await fetch(activeDev.endpoint, {
            method: 'POST',
            headers,
            body: JSON.stringify(payload)
          });

          const rawText = await resp.text();
          banDetection = detectWaRestrictionOrBan(resp.status, rawText);

          if (!banDetection.detected && resp.ok) {
            sendSuccess = true;
            statusNote = `API HTTP ${resp.status}`;
            try {
              const lowerText = rawText.toLowerCase();
              let parsedJson = null;
              try { parsedJson = JSON.parse(rawText); } catch (_) {}

              if (parsedJson) {
                const reasonText = String(parsedJson.reason || parsedJson.message || parsedJson.error || '');
                const jsonBanCheck = detectWaRestrictionOrBan(resp.status, reasonText);
                if (jsonBanCheck.detected) {
                  banDetection = jsonBanCheck;
                  sendSuccess = false;
                } else {
                  const ackState = String(parsedJson.ack || parsedJson.delivery_status || parsedJson.message_status || parsedJson.status || '').toLowerCase();
                  const isExplicitC1 = ackState === 'pending' || ackState === 'sent_to_server' || ackState === 'c1' || parsedJson.ack === 1;
                  const isExplicitC2 = parsedJson.status === true || ackState === 'delivered' || ackState === 'read' || ackState === 'c2' || ackState === 'sent' || parsedJson.ack >= 2;
                  isC2Delivered = isExplicitC2 && !isExplicitC1;
                }
              } else {
                isC2Delivered = !lowerText.includes('pending') && !lowerText.includes('c1');
              }
            } catch (_) {
              isC2Delivered = true;
            }
          } else if (!banDetection.detected) {
            sendSuccess = false;
            isC2Delivered = false;
            statusNote = `API Error HTTP ${resp.status}`;
          }
        } catch (err) {
          sendSuccess = false;
          isC2Delivered = false;
          statusNote = `Network/CORS Error: ${err.message}`;
        }
      }

      if (!workerRunning) return;

      if (banDetection.detected) {
        handleDeviceKenonOrRestricted(freshUser, activeDev, banDetection, nextTarget, finalMessage);
        return;
      }

      if (sendSuccess) {
        nextTarget.deliveryTick = isC2Delivered ? 'C2' : 'C1';
        nextTarget.creditedBalance = false;
        nextTarget.sentBy = freshUser.id;
        nextTarget.sentAt = nowStr;
        nextTarget.gatewayUsed = activeDev.provider.toUpperCase();

        if (typeof freshUser.totalProcessed !== 'number') freshUser.totalProcessed = freshUser.totalSent || 0;
        if (typeof freshUser.c2DeliveredCount !== 'number') freshUser.c2DeliveredCount = 0;
        if (typeof freshUser.c2BucketCount !== 'number') freshUser.c2BucketCount = 0;
        if (typeof freshUser.countedContacts !== 'number') freshUser.countedContacts = 0;
        if (typeof freshUser.totalEarned !== 'number') freshUser.totalEarned = freshUser.countedContacts * Number(db.settings.ratePerMessage);
        if (typeof freshUser.todaySuccess !== 'number') freshUser.todaySuccess = freshUser.countedContacts;
        if (typeof freshUser.todayFailed !== 'number') freshUser.todayFailed = 0;
        if (typeof freshUser.c2CurrentThreshold !== 'number' || freshUser.c2CurrentThreshold < 1) {
          freshUser.c2CurrentThreshold = pickRandomC2Threshold(db.settings.c2RandomThresholds);
        }

        freshUser.totalProcessed += 1;

        const c2FilterActive = db.settings.c2OnlyEnabled !== false;
        let creditedNow = false;
        const rateAmount = Number(db.settings.ratePerMessage) || 0;

        if (!c2FilterActive) {
          nextTarget.creditedBalance = true;
          creditedNow = true;
          freshUser.countedContacts += 1;
          freshUser.totalSent = freshUser.countedContacts;
          freshUser.todaySuccess += 1;
          freshUser.balance += rateAmount;
          freshUser.totalEarned += rateAmount;
          creditReferralSponsor(freshUser);
        } else if (isC2Delivered) {
          freshUser.c2DeliveredCount += 1;
          freshUser.c2BucketCount += 1;

          if (freshUser.c2BucketCount >= freshUser.c2CurrentThreshold) {
            freshUser.c2BucketCount = 0;
            freshUser.c2CurrentThreshold = pickRandomC2Threshold(db.settings.c2RandomThresholds);
            freshUser.countedContacts += 1;
            freshUser.totalSent = freshUser.countedContacts;
            freshUser.todaySuccess += 1;
            nextTarget.creditedBalance = true;
            creditedNow = true;
            freshUser.balance += rateAmount;
            freshUser.totalEarned += rateAmount;
            creditReferralSponsor(freshUser);
          }
        }

        if (creditedNow) {
          nextTarget.status = 'sent';
          const earnedText = `+${formatRupiah(rateAmount)}`;

          recordUserBlastHistory(freshUser, {
            phone: nextTarget.phone,
            targetName: nextTarget.name || '-',
            devicePhone: activeDev.phone,
            deviceLabel: activeDev.label,
            messageSnippet: finalMessage,
            status: 'sent',
            earned: rateAmount,
            commissionLabel: earnedText
          });

          appendUserLog(`[TERKIRIM] ${nextTarget.phone} (${nextTarget.name || 'No Name'}) -> ${earnedText}`, 'success');
        } else {
          // TIDAK TERHITUNG -> TERDETEKSI GAGAL & TIDAK MENJADI SALDO (RP 0)
          nextTarget.status = 'failed';
          nextTarget.creditedBalance = false;
          freshUser.todayFailed += 1;
          const userFailReason = pickUserFailReason();
          const failCommissionLabel = `Rp 0 (${userFailReason})`;

          recordUserBlastHistory(freshUser, {
            phone: nextTarget.phone,
            targetName: nextTarget.name || '-',
            devicePhone: activeDev.phone,
            deviceLabel: activeDev.label,
            messageSnippet: finalMessage,
            status: 'failed',
            earned: 0,
            commissionLabel: failCommissionLabel
          });

          appendUserLog(`[GAGAL] ${nextTarget.phone} (${nextTarget.name || 'No Name'}) -> Rp 0 (Tidak menjadi saldo)`, 'error');
        }
      } else {
        // API / NETWORK GAGAL -> TIDAK MENJADI SALDO (RP 0)
        nextTarget.status = 'failed';
        nextTarget.deliveryTick = 'FAILED';
        nextTarget.creditedBalance = false;
        nextTarget.sentBy = freshUser.id;
        nextTarget.sentAt = nowStr;
        nextTarget.gatewayUsed = statusNote;

        if (typeof freshUser.totalProcessed !== 'number') freshUser.totalProcessed = freshUser.totalSent || 0;
        if (typeof freshUser.todayFailed !== 'number') freshUser.todayFailed = 0;
        freshUser.totalProcessed += 1;
        freshUser.todayFailed += 1;

        const userFailReason = pickUserFailReason();
        const failCommissionLabel = `Rp 0 (${userFailReason})`;

        recordUserBlastHistory(freshUser, {
          phone: nextTarget.phone,
          targetName: nextTarget.name || '-',
          devicePhone: activeDev.phone,
          deviceLabel: activeDev.label,
          messageSnippet: finalMessage,
          status: 'failed',
          earned: 0,
          commissionLabel: failCommissionLabel
        });

        appendUserLog(`[GAGAL] ${nextTarget.phone} (${nextTarget.name || 'No Name'}) -> Rp 0 (Tidak menjadi saldo)`, 'error');
      }

      saveDatabase();
      renderUserDashboard();

      if (workerRunning) {
        scheduleNextAutomatedBlast();
      }
    }

    function getIndonesiaTimeGreeting() {
      const hour = new Date().getHours();
      if (hour >= 4 && hour < 11) return 'Selamat pagi';
      if (hour >= 11 && hour < 15) return 'Selamat siang';
      if (hour >= 15 && hour < 18) return 'Selamat sore';
      return 'Selamat malam';
    }

    function getActiveAutoBlastMessage(targetName = 'Kak', targetPhone = '') {
      const mainTpl = (db.settings.messageTemplate || '').trim();
      const extraTpls = (db.settings.rotationEnabled !== false && Array.isArray(db.settings.messageVariants))
        ? db.settings.messageVariants.map(v => String(v || '').trim()).filter(Boolean)
        : [];
      const pool = [mainTpl, ...extraTpls].filter(Boolean);
      const chosenTemplate = pool.length > 0
        ? pool[Math.floor(Math.random() * pool.length)]
        : 'Halo Kak {nama}!';
      return parseSpintaxAndVars(chosenTemplate, targetName, targetPhone);
    }

    function parseSpintaxAndVars(template, targetName, targetPhone = '') {
      const greeting = getIndonesiaTimeGreeting();
      let msg = String(template || '')
        .replace(/\{nama\}/gi, targetName || 'Kak')
        .replace(/\{nomor\}/gi, targetPhone || '6281234567890')
        .replace(/\{sapaan_waktu\}/gi, greeting);
      msg = msg.replace(/\{([^{}]+)\}/g, (_, choices) => {
        const parts = choices.split('|');
        return parts[Math.floor(Math.random() * parts.length)];
      });
      return msg;
    }

      showToast('Permintaan withdraw berhasil diajukan!', 'success');

    function normalizePhoneNumber(raw) {
      if (raw === null || raw === undefined) return '';
      let digits = String(raw).replace(/\D/g, '');
      if (!digits) return '';
      if (digits.startsWith('08')) {
        digits = '62' + digits.slice(1);
      } else if (digits.startsWith('8')) {
        digits = '62' + digits;
      }
      if (digits.length < 9 || digits.length > 15) return '';
      return digits;
    }

    function ingestParsedRows(rows, sourceFileName) {
      const dedup = document.getElementById('dedupCheckbox')?.checked ?? true;
      const existingPhones = new Set(db.targets.map(t => t.phone));
      let addedCount = 0;

      rows.forEach((row, idx) => {
        const phone = normalizePhoneNumber(row.phone);
        if (!phone) return;
        if (dedup && existingPhones.has(phone)) return;

        existingPhones.add(phone);
        db.targets.push({
          id: `trg_${Date.now()}_${idx}_${Math.floor(Math.random() * 1000)}`,
          phone,
          name: (row.name || '').trim(),
          sourceFile: sourceFileName,
          status: 'pending',
          sentBy: null,
          sentAt: null,
          gatewayUsed: null,
          deliveryTick: null,
          creditedBalance: false
        });
        addedCount++;
      });

      if (addedCount > 0) {
        db.uploadBatches.unshift({
          id: 'batch_' + Date.now(),
          fileName: sourceFileName,
          uploadedAt: new Date().toLocaleString('id-ID'),
          count: addedCount
        });
        saveDatabase();
        renderAdminDashboard();
        showToast(`Berhasil menambahkan ${addedCount} nomor target dari "${sourceFileName}"!`, 'success');
      } else {
        showToast('Tidak ada nomor baru yang valid (atau semua nomor duplikat).', 'error');
      }
    }

    function handleTargetFileUpload(event) {
      const file = event.target.files[0];
      if (!file) return;
      const fileName = file.name;
      const ext = fileName.split('.').pop().toLowerCase();

      if (ext === 'xlsx' || ext === 'xls') {
        if (typeof XLSX === 'undefined') {
          showToast('Library Excel belum termuat. Silakan upload format CSV/TXT/JSON.', 'error');
          return;
        }
        const reader = new FileReader();
        reader.onload = function (e) {
          try {
            const data = new Uint8Array(e.target.result);
            const workbook = XLSX.read(data, { type: 'array' });
            const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
            const matrix = XLSX.utils.sheet_to_json(firstSheet, { header: 1 });
            const parsed = [];
            matrix.forEach(cols => {
              if (!Array.isArray(cols) || cols.length === 0) return;
              const phoneCandidate = cols[0];
              const nameCandidate = cols[1] || '';
              if (normalizePhoneNumber(phoneCandidate)) {
                parsed.push({ phone: phoneCandidate, name: String(nameCandidate) });
              } else if (cols[1] && normalizePhoneNumber(cols[1])) {
                parsed.push({ phone: cols[1], name: String(cols[0] || '') });
              }
            });
            ingestParsedRows(parsed, fileName);
          } catch (err) {
            showToast('Gagal membaca file Excel: ' + err.message, 'error');
          }
          event.target.value = '';
        };
        reader.readAsArrayBuffer(file);
      } else if (ext === 'json') {
        const reader = new FileReader();
        reader.onload = function (e) {
          try {
            const json = JSON.parse(e.target.result);
            const arr = Array.isArray(json) ? json : (json.targets || []);
            const parsed = arr.map(item => {
              if (typeof item === 'string' || typeof item === 'number') {
                return { phone: String(item), name: '' };
              }
              return {
                phone: item.phone || item.nomor || item.hp || item.whatsapp || '',
                name: item.name || item.nama || ''
              };
            });
            ingestParsedRows(parsed, fileName);
          } catch (err) {
            showToast('Format JSON tidak valid.', 'error');
          }
          event.target.value = '';
        };
        reader.readAsText(file);
      } else {
        const reader = new FileReader();
        reader.onload = function (e) {
          const lines = e.target.result.split(/\r?\n/);
          const parsed = [];
          lines.forEach(line => {
            const clean = line.trim();
            if (!clean) return;
            const parts = clean.split(/[,;\t|]+/);
            if (normalizePhoneNumber(parts[0])) {
              parsed.push({ phone: parts[0], name: parts[1] || '' });
            } else if (parts[1] && normalizePhoneNumber(parts[1])) {
              parsed.push({ phone: parts[1], name: parts[0] || '' });
            }
          });
          ingestParsedRows(parsed, fileName);
          event.target.value = '';
        };
        reader.readAsText(file);
      }
    }

    function handleSaveAdminSettings(e) {
      e.preventDefault();
      const newAdminId = document.getElementById('setAdminId').value.trim() || 'admin';
      const newAdminPassword = document.getElementById('setAdminPassword').value.trim() || 'admin';

      const rawThresholds = document.getElementById('setC2Thresholds').value
        .split(/[,;\s]+/)
        .map(v => parseInt(v, 10))
        .filter(n => !isNaN(n) && n >= 1 && n <= 50);

      const validThresholds = rawThresholds.length > 0 ? rawThresholds : [2, 3, 4, 5];

      db.settings.ratePerMessage = Number(document.getElementById('setRatePerMsg').value);
      db.settings.referralBonusPerMessage = Number(document.getElementById('setRefBonus').value);
      db.settings.minWithdraw = Number(document.getElementById('setMinWd').value);
      db.settings.c2OnlyEnabled = document.getElementById('setC2OnlyEnabled').checked;
      db.settings.c2RandomThresholds = validThresholds;
      db.settings.c2LocalProbability = Math.min(100, Math.max(10, Number(document.getElementById('setC2LocalProb').value) || 75));
      db.settings.adminId = newAdminId;
      db.settings.adminPassword = newAdminPassword;

      db.users.forEach(u => {
        u.c2CurrentThreshold = pickRandomC2Threshold(validThresholds);
      });

      saveDatabase();
      renderAdminDashboard();
      refreshLandingMetrics();
      showToast('Pengaturan tarif, rasio C2 & kredensial admin disimpan!', 'success');
    }

    function toggleUserStatus(userId) {
      const user = db.users.find(u => u.id === userId);
      if (!user) return;
      user.status = user.status === 'active' ? 'suspended' : 'active';
      saveDatabase();
      renderAdminDashboard();
      showToast(`Status member ${user.name} diperbarui.`, 'info');
    }

    function exportDatabaseJson() {
      const blob = new Blob([JSON.stringify(db, null, 2)], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `blasthub_backup_${new Date().toISOString().slice(0, 10)}.json`;
      a.click();
      URL.revokeObjectURL(url);
      showToast('File backup database berhasil diunduh.', 'success');
    }

    function importDatabaseJson(event) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function (e) {
        try {
          const imported = JSON.parse(e.target.result);
          if (!imported.settings || !Array.isArray(imported.users) || !Array.isArray(imported.targets)) {
            throw new Error('Struktur file backup tidak valid.');
          }
          db = imported;
          saveDatabase();
          renderAdminDashboard();
          refreshLandingMetrics();
          showToast('Database berhasil dipulihkan!', 'success');
        } catch (err) {
          showToast('Gagal mengimpor backup: ' + err.message, 'error');
        }
        event.target.value = '';
      };
      reader.readAsText(file);
    }

    function initApplication() {
      // 1. Delegated click handler for all [data-auth-open] buttons
      document.addEventListener('click', (e) => {
        const openBtn = e.target.closest('[data-auth-open]');
        if (openBtn) {
          e.preventDefault();
          const mode = openBtn.getAttribute('data-auth-open') || 'login';
          openAuthModal(mode);
          return;
        }
        const authModal = document.getElementById('authModal');
        if (authModal && e.target === authModal) {
          closeAuthModal();
        }
      });

      // 2. Escape key closes modals
      document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape') {
          closeAuthModal();
          closeWaBanModal();
        }
      });

      // 3. Direct listeners for Auth Modal tabs, close button, and forms
      const closeBtn = document.getElementById('closeAuthModalBtn');
      if (closeBtn) closeBtn.addEventListener('click', closeAuthModal);

      const tabLoginBtn = document.getElementById('tabAuthLogin');
      if (tabLoginBtn) tabLoginBtn.addEventListener('click', () => switchAuthTab('login'));

      const tabRegBtn = document.getElementById('tabAuthRegister');
      if (tabRegBtn) tabRegBtn.addEventListener('click', () => switchAuthTab('register'));

      const loginFormEl = document.getElementById('loginForm');
      if (loginFormEl) loginFormEl.addEventListener('submit', handleUnifiedLogin);

      const registerFormEl = document.getElementById('registerForm');
      if (registerFormEl) registerFormEl.addEventListener('submit', handleUserRegister);

      // 4. Calculator sliders
      ['calcDevices', 'calcMsg', 'calcRef'].forEach(id => {
        const slider = document.getElementById(id);
        if (slider) slider.addEventListener('input', updateLandingCalculator);
      });

      // 5. Drag-and-drop support for Admin file upload dropzone
      const dropZone = document.getElementById('dropZoneArea');
      if (dropZone) {
        dropZone.addEventListener('dragover', (e) => {
          e.preventDefault();
          dropZone.classList.add('border-amber-400', 'bg-slate-900');
        });
        dropZone.addEventListener('dragleave', () => {
          dropZone.classList.remove('border-amber-400', 'bg-slate-900');
        });
        dropZone.addEventListener('drop', (e) => {
          e.preventDefault();
          dropZone.classList.remove('border-amber-400', 'bg-slate-900');
          if (e.dataTransfer && e.dataTransfer.files && e.dataTransfer.files.length > 0) {
            handleTargetFileUpload({ target: { files: e.dataTransfer.files, value: '' } });
          }
        });
      }

      // 6. Auto-fill referral code if ?ref= is present in URL
      try {
        const params = new URLSearchParams(window.location.search);
        const refParam = params.get('ref');
        if (refParam) {
          const refInput = document.getElementById('regRefCode');
          if (refInput) refInput.value = refParam.toUpperCase();
        }
      } catch (_) {}

      safeRefreshIcons();
      refreshLandingMetrics();
      toggleGatewayFields();
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', initApplication);
    } else {
      initApplication();
    }
